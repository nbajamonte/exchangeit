function getSaved(){ return readStore(KEYS.saved,[]); }
function findSource(id){ for(const [cat,list] of Object.entries(SOURCE_CATALOG)){ const s=list.find(x=>x.id===id); if(s) return {...s,category:cat}; } return null; }
function isSaved(id){ return getSaved().some(x=>x.id===id); }
function toggleSaved(id,query,category,button){
  let saved=getSaved(); const index=saved.findIndex(x=>x.id===id);
  if(index>=0){ saved.splice(index,1); toast('Fonte rimossa dai salvati'); }
  else { const source=findSource(id); if(source) saved.unshift({id,query,category,url:buildUrl(source,query),name:source.name,access:source.access,kind:source.kind}); toast('Fonte salvata'); }
  writeStore(KEYS.saved,saved); updateSavedCount(); renderSaved();
  if(button){ const active=isSaved(id); button.classList.toggle('is-saved',active); button.textContent=active?'✓ Salvata':'＋ Salva'; }
}
