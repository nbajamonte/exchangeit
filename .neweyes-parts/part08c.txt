function renderSaved(){
  const box=document.getElementById('savedList'); const saved=getSaved();
  if(!saved.length){ box.innerHTML='<div class="saved-empty">Non hai ancora salvato nessuna fonte.</div>'; return; }
  box.innerHTML=saved.map((x,i)=>`<div class="saved-item"><div><b>${escapeHtml(x.name)}</b><p>${escapeHtml(x.kind)} · ${escapeHtml(x.access)}${x.query?' · “'+escapeHtml(x.query)+'”':''}</p></div><div class="saved-item-actions"><a href="${x.url}" target="_blank" rel="noopener noreferrer">Apri ↗</a><button type="button" data-remove-saved="${i}">Rimuovi</button></div></div>`).join('');
  box.querySelectorAll('[data-remove-saved]').forEach(btn=>btn.addEventListener('click',()=>{ const next=getSaved(); next.splice(Number(btn.dataset.removeSaved),1); writeStore(KEYS.saved,next); renderSaved(); updateSavedCount(); if(currentQuery) showResult(currentQuery,classify(currentQuery)); }));
}
